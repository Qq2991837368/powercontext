"""Mini catalogue service: book listing filter.

The module is intentionally small so that the SDD workflow can show a complete
spec -> context -> code-change -> verification loop against a real database.
"""

from __future__ import annotations

from collections.abc import Iterable, Mapping
from typing import Any

__all__ = ["filter_books"]


def _is_number(value: Any) -> bool:
    """Return ``True`` for real numeric values (booleans are not prices)."""
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def _as_tag_list(value: Any) -> list[Any]:
    """Normalise a tags container into a plain list of tag values."""
    if value is None:
        return []
    if isinstance(value, str):
        return [value]
    if isinstance(value, (list, tuple, set, frozenset)):
        return list(value)
    try:
        return list(value)
    except TypeError:
        return [value]


def filter_books(
    books: Iterable[Mapping[str, Any]],
    *,
    keyword: str | None = None,
    max_price: float | None = None,
    tags: Iterable[str] | None = None,
    include_out_of_stock: bool = False,
) -> list[dict[str, Any]]:
    """Filter catalogue books according to spec v1 (R1-R4).

    * R1: ``keyword`` matches ``title`` case-insensitively; ``None``/empty
      disables the filter.
    * R2: ``max_price`` keeps books with ``price <= max_price`` (inclusive);
      books without a numeric price are dropped when the filter is enabled.
    * R3: ``tags`` uses all-of semantics; books lacking a ``tags`` field are
      dropped when the filter is enabled.
    * R4: books explicitly marked ``in_stock=False`` are excluded unless
      ``include_out_of_stock=True`` is passed.

    The input objects are never mutated: every returned row is a shallow copy
    and the input ordering is preserved.
    """
    needle = (keyword or "").strip().lower()
    requested_tags = _as_tag_list(tags) if tags is not None else []
    filter_by_price = max_price is not None
    filter_by_tags = bool(requested_tags)

    selected: list[dict[str, Any]] = []
    for book in books:
        title = str(book.get("title", ""))
        if needle and needle not in title.lower():
            continue

        if filter_by_price:
            price = book.get("price")
            if not _is_number(price) or price > max_price:
                continue

        if filter_by_tags:
            if "tags" not in book:
                continue
            book_tags = _as_tag_list(book.get("tags"))
            if not all(tag in book_tags for tag in requested_tags):
                continue

        if not include_out_of_stock and book.get("in_stock", True) is False:
            continue

        selected.append(dict(book))
    return selected

# 图书过滤器规格（Book Filter Specification）

Version: v2
Status: active
Owner: catalogue-team

## R1 关键词匹配

`filter_books(books, keyword=...)` 返回 `title` 中包含关键词的图书，大小写不敏感；
`keyword` 为 `None` 或空字符串时不做过滤。结果保持输入顺序。

## R2 最高价格（包含边界）

传入 `max_price` 时，仅保留 `price <= max_price` 的图书，边界值必须包含。
`price` 缺失或不是数字的图书在启用该过滤时一律排除。

## R3 标签全匹配

传入 `tags` 时，图书必须包含全部请求标签（all-of 语义）。
`tags` 为 `None` 或空序列时不做过滤；缺少 `tags` 字段的图书在启用该过滤时排除。

## R4 库存可见性

默认排除 `in_stock` 显式为 `False` 的图书；`in_stock` 缺失视为有货。
调用方传入 `include_out_of_stock=True` 时保留缺货图书。

## R5 最低价格（包含边界，v2 新增）

传入 `min_price` 时，仅保留 `price >= min_price` 的图书，边界值必须包含。
`price` 缺失或不是数字的图书在启用该过滤时一律排除。
`min_price` 与 `max_price` 同时存在时构成闭区间 `[min_price, max_price]`，
闭区间内没有图书时返回空列表。

## N1 通用约束

- 不修改输入对象，返回结果为浅拷贝（保留调用方附加字段）。
- 结果顺序与输入顺序一致。
- 不修改测试与规格文件来“适配”实现；实现必须满足上述规则。

"""Target tests for spec v2 requirement R5 (min_price)."""

from src.book_filter import filter_books

BOOKS = [
    {"id": "b1", "title": "Clean Code", "price": 45.0, "tags": ["engineering", "classic"], "in_stock": True},
    {"id": "b2", "title": "Clean Architecture", "price": 59.0, "tags": ["engineering"], "in_stock": True},
    {"id": "b3", "title": "Refactoring", "price": 60.0, "tags": ["engineering", "classic"], "in_stock": False},
    {"id": "b4", "title": "Dune", "price": 30.0, "tags": ["fiction"], "in_stock": True},
    {"id": "b5", "title": "Untitled Draft", "tags": ["engineering"], "in_stock": True},
    {"id": "b6", "title": "Legacy Notes", "price": 12.0, "tags": ["reference"], "promoted": True},
]


def ids(rows):
    return [row["id"] for row in rows]


def test_min_price_is_inclusive_and_excludes_missing_price():
    rows = filter_books(BOOKS, min_price=45.0, include_out_of_stock=True)
    assert ids(rows) == ["b1", "b2", "b3"]


def test_price_range_is_a_closed_interval():
    rows = filter_books(BOOKS, min_price=45.0, max_price=59.0, include_out_of_stock=True)
    assert ids(rows) == ["b1", "b2"]


def test_empty_price_range_returns_no_rows():
    rows = filter_books(BOOKS, min_price=61.0, max_price=70.0, include_out_of_stock=True)
    assert rows == []
